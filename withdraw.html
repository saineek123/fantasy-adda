function handleWithdrawAction(id, action, note){ db.ref('withdrawals/'+id).once('value').then(s=>{ const req = s.val(); if(!req){ toast('Request not found','error'); return; } const updates = {}; updates['withdrawals/'+id+'/status'] = action === 'approve' ? 'approved' : 'rejected'; updates['withdrawals/'+id+'/processedAt'] = Date.now(); updates['withdrawals/'+id+'/processedBy'] = state.uid; if(note) updates['withdrawals/'+id+'/adminNote'] = note; const amt = Number(req.amount||0); if(action === 'approve'){ /* Bal already deduct ho chuka hai user side par. Sirf status approve karo, aur tx me approved mark karo. */ updates['withdrawals/'+id+'/locked'] = false; const txKey = db.ref('transactions').push().key; updates['transactions/'+txKey] = { userId: req.userId, type: 'debit', amount: amt, description: 'Withdrawal approved by admin (Winning) ' + (note?('('+note+')'):''), date: Date.now(), requestId: id, source: 'winning', status: 'approved' }; db.ref().update(updates) .then(()=>{ toast('Withdrawal approved','success'); closeModal('noteModal'); loadWithdrawals(); }) .catch(e=>toast(e.message,'error')); }else{ /* REJECT → refund winning + balance back */ db.ref('users/'+req.userId+'/wallet').transaction(w=>{ if(!w) w = {balance:0, winning:0, deposit:0}; const win = Number(w.winning||0); const bal = Number(w.balance||0); w.winning = Number((win + amt).toFixed(2)); w.balance = Number((bal + amt).toFixed(2)); return w; }).then(()=>{ const txKey = db.ref('transactions').push().key; updates['transactions/'+txKey] = { userId: req.userId, type: 'credit', amount: amt, description: 'Withdrawal rejected - refunded to winning ' + (note?('('+note+')'):''), date: Date.now(), requestId: id, source: 'winning', status: 'refunded' }; db.ref().update(updates) .then(()=>{ toast('Withdrawal rejected & refunded','success'); closeModal('noteModal'); loadWithdrawals(); }) .catch(e=>toast(e.message,'error')); }).catch(e=>toast(e.message,'error')); } }); }
